3. Builds & Deployments

Trigger a Build (Binary S2I)

# From repo root — uploads local directory as build context
oc start-build <bc-name> -n <namespace> --from-dir=platform/<product>/<dir>/ --follow

Base Images (internal registry — no internet pull needed)

Base ImageInternal Tag
node:20-alpineimage-registry.openshift-image-registry.svc:5000/openshift/node:20-alpine
python:3.11-slimimage-registry.openshift-image-registry.svc:5000/openshift/python:3.11-slim

Apply a YAML Change (GitOps — preferred)

git add platform/<file>.yaml
git commit -m "fix: description"
git push origin main
# ArgoCD auto-applies within ~3 min

Manual Apply (immediate)

oc apply -f platform/<namespace>/<file>.yaml

Rolling Restart

oc rollout restart deployment/<name> -n <namespace>

RWO PVC Multi-Attach Fix (most common issue)

# Old pod holds RWO volume — new pod stuck ContainerCreating
oc delete pod <old-pod> -n <ns> --grace-period=0 --force
# If old ReplicaSet keeps respawning:
oc scale rs <old-rs> --replicas=0 -n <ns>

4. Secret Management

Patch Pattern (never commit real secrets to git without bypass)

# Write patch file (stringData = plain text, auto base64-encoded)
echo '{"stringData":{"KEY":"value"}}' > /tmp/patch.json
oc patch secret <secret-name> -n <ns> --type merge --patch-file /tmp/patch.json

Admin REST API — Get Token

ADMIN_PW=$(oc get secret credential-i3-keycloak -n i3-auth \
  -o jsonpath='{.data.ADMIN_PASSWORD}' | base64 -d)

oc exec -n i3-auth keycloak-0 -c keycloak -- bash -c "
  curl -sf -X POST http://localhost:8080/auth/realms/master/protocol/openid-connect/token \
    -d 'client_id=admin-cli&grant_type=password&username=admin&password=$ADMIN_PW' \
    -H 'Content-Type: application/x-www-form-urlencoded' \
  | python3 -c 'import sys,json; print(json.load(sys.stdin)[\"access_token\"])'
"

Create / Reset a User (via script)

# Use platform/scripts/ensure_evalos_users.sh as template
oc cp platform/scripts/ensure_evalos_users.sh i3-auth/keycloak-0:/tmp/script.sh -c keycloak
oc exec -n i3-auth keycloak-0 -c keycloak -- bash /tmp/script.sh "$ADMIN_PW"

Rotate a Client Secret

# Inside keycloak pod:
TOKEN=... # (get token as above)
CID=$(curl -sf -H "Authorization: Bearer $TOKEN" \
  "http://localhost:8080/auth/admin/realms/i3/clients?clientId=<clientId>" \
  | python3 -c 'import sys,json; print(json.load(sys.stdin)[0]["id"])')
NEW_SECRET=$(curl -sf -X POST -H "Authorization: Bearer $TOKEN" \
  "http://localhost:8080/auth/admin/realms/i3/clients/$CID/client-secret" \
  | python3 -c 'import sys,json; print(json.load(sys.stdin)["value"])')
echo "New secret: $NEW_SECRET"
# Then patch the cluster secret:
oc patch secret <product>-secrets -n <ns> --type merge \
  -p "{\"stringData\":{\"KEYCLOAK_CLIENT_SECRET\":\"$NEW_SECRET\"}}"

9. Troubleshooting

🔴 RWO PVC Multi-Attach — Pod stuck ContainerCreating

Symptom: FailedAttachVolume: Multi-Attach error for volume ... already used by pod(s) X

oc delete pod <old-pod> -n <ns> --grace-period=0 --force
oc scale rs <old-rs> --replicas=0 -n <ns> # stop RS respawning it

🔴 NetworkPolicy DNS failure — Pod can't resolve internal services

Symptom: Pods with Egress NetworkPolicy fail DNS lookups. Default port 53 rule without to: doesn't work in OVN.

# Fix: Add explicit DNS egress to openshift-dns namespace
egress:
- to:
    - namespaceSelector:
        matchLabels:
          kubernetes.io/metadata.name: openshift-dns
    ports:
      - protocol: UDP
        port: 5353

🔴 Open WebUI crashes — HuggingFace model download fails (no internet)

Symptom: LocalEntryNotFoundError: ConnectError: [Errno -3] Temporary failure in name resolution

# Fix: Set Ollama as embedding engine in the ConfigMap
oc patch cm open-webui-<name>-config -n <ns> --type merge \
  -p '{"data":{"RAG_EMBEDDING_ENGINE":"ollama","RAG_EMBEDDING_MODEL":"nomic-embed-text:latest","OLLAMA_BASE_URL":"http://ollama.i3-model-gateway.svc.cluster.local:11434","ENABLE_OLLAMA_API":"true"}}'
# Then reset the DB (the config is stored in webui.db):
oc scale deployment <name> -n <ns> --replicas=0
# Create reset pod, delete webui.db, scale back to 1

🔴 ChromaDB crashes — "Unable to configure handler 'file'"

Symptom: ChromaDB pod CrashLoopBackOff with uvicorn log config error.

# Fix: Override uvicorn cmd with custom stdout-only log-config
command: ["uvicorn","chromadb.app:app","--workers=1","--host=0.0.0.0","--port=8000","--proxy-headers","--log-config=/etc/chromadb/log_config.yml"]
# Mount a ConfigMap with stdout-only log_config.yml (see platform/ai-lab/chromadb/chromadb-deploy.yaml)

🔴 Build fails — "must specify limits.cpu/memory for: <container>"

Symptom: Pod creation forbidden by ResourceQuota.

# Fix: Patch BuildConfig with resource limits
oc patch bc <bc-name> -n <ns> --type merge --patch-file platform/scripts/bc-resources-patch.json

🔴 ERPNext permission denied creating symlink in /sites

Symptom: ln: failed to create symbolic link: Permission denied

# Fix: Create ServiceAccount with anyuid SCC
oc create sa erpnext-sa -n i3-afroerp
oc adm policy add-scc-to-user anyuid -z erpnext-sa -n i3-afroerp
# Set serviceAccountName: erpnext-sa in deployment

🔴 JupyterHub "configurable-http-proxy not found"

Symptom: FileNotFoundError: 'configurable-http-proxy' — k8s-hub image doesn't include the proxy.

# Fix: Add proxy sidecar container to the deployment
image: jupyterhub/configurable-http-proxy:4
command: [configurable-http-proxy, --port=8000, --api-ip=127.0.0.1, --api-port=8001, ...]
# In jupyterhub_config.py:
c.ConfigurableHTTPProxy.should_start = False
c.ConfigurableHTTPProxy.api_url = "http://127.0.0.1:8001"

11. Disaster Recovery

Recovery Priority Order

  1. Postgres HA (i3-data) — data layer first
  2. Keycloak (i3-auth) — authentication
  3. Model Gateway — Ollama + LiteLLM
  4. EvalOS (revenue-critical)
  5. Remaining product namespaces

Re-bootstrap from GitOps

# 1. Login to new/restored cluster
oc login https://<new-cluster-api> --token=<token>
# 2. Apply all namespaces + base infra
oc apply -f platform/ --recursive
# 3. Re-patch all secrets (see platform/scripts/ patch files)
# 4. Trigger builds for all products
# 5. Re-pull Ollama models (takes ~30 min for all 6)
oc exec -n i3-model-gateway ollama-0 -c ollama -- ollama pull qwen2.5:7b-instruct-q4_K_M
# ... repeat for all 6 models

Keycloak Realm Export

oc exec -n i3-auth keycloak-0 -c keycloak -- \
  /opt/keycloak/bin/kc.sh export --realm i3 --file /tmp/i3-realm-export.json
oc cp i3-auth/keycloak-0:/tmp/i3-realm-export.json ./i3-realm-backup-$(date +%Y%m%d).json -c keycloak

Postgres Backup (manual)

# PGO handles automated backups to repo1. Manual dump:
oc run pg-dump --image=postgres:15 -n i3-data --restart=Never \
  --env="PGPASSWORD=<pw>" \
  --command -- pg_dump -h i3-postgres-pgbouncer.i3-data.svc \
  -U <user> -d <db> -f /tmp/backup.sql
